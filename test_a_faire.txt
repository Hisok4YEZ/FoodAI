Ce test est parfait pour montrer au prof que tout marche :

for c in caesar_salad chocolate_cake fried_rice hamburger ice_cream omelette pizza ramen spaghetti_bolognese sushi; do

  echo "==================== $c | $img ===================="
  python -m app.week4.predict_to_recipe --image "data/food101_10/test/$c/$img" --servings 2 --min_conf 0.70
done
